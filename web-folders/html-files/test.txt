function FilterProject(ARR,category, location, Years){
   
   const filterProjects = ARR.filter(project => {
    return (
        ( category === "All" || project.type.toLowerCase().includes(category.toLowerCase())) &&
        ( location === "All" || project.location.toLowerCase().includes(location.toLowerCase())) &&
        (Years === "All" || project.year === Number(Years))
    );
   })

   return filterProjects

}
